\begin{frame}{The exposure we must measure (Capitolis kickoff definition)}
  \[ \max\big(V(t{+}10\text{bd}) - V(t{-}1\text{bd}),\,0\big) \]
  \begin{itemize}
    \item \textbf{Variation margin:} netted value one business day before $t$, two-way, no threshold
    \item \textbf{Margin period of risk (MPoR):} close-out 10 business days after $t$
    \item \textbf{Netting:} within a counterparty only -- book exposure = sum over netting sets; \emph{never} netted across counterparties
    \item \textbf{Trades settling inside the window} are excluded from both legs (a scheduled settlement is cash, not a market move)
  \end{itemize}

\end{frame}

\partframe{IV}{Efficient Simulation}
\section{Part IV: Simulation Methodology}

\begin{frame}{Sampling scheme: five methods, exact benchmark numbers}
  \small
  \textbf{Controlled test (European call, known Black-Scholes truth = 41.077):}
  \begin{tabular}{lrrr}
    \toprule
    \textbf{Method} & \textbf{RMSE} & \textbf{vs.\ pseudo-random} & \textbf{ms/trial} \\
    \midrule
    Pseudo-random & 1.456 & 1.0x (baseline) & 0.30 \\
    Antithetic & 1.079 & 1.3x better & 0.31 \\
    Moment-matched & 0.232 & 6.3x better & 0.55 \\
    Sobol & 0.051 & 28.8x better & 1.65 \\
    \textbf{Latin Hypercube} & \textbf{0.029} & \textbf{50.6x better} & 0.98 \\
    \bottomrule
  \end{tabular}
  \vspace{6pt}

  \textbf{Real 663-dimensional engine check (std.\ dev.\ of PFE / median, across repeats):} no method wins both -- Sobol has the lowest PFE std (2.72M vs.\ 4.29M pseudo-random), but moment-matched has the lowest median std (0.40M vs.\ 0.76M pseudo-random); Latin Hypercube is competitive on both (3.73M / 0.48M) without being worst on either.
  \vspace{4pt}
  \good\ Latin Hypercube: best on the controlled test, competitive (never worst) on both real-engine statistics, negligible extra cost.
\end{frame}

\begin{frame}{Latin Hypercube: what it does and does not do}
  \small
  \begin{itemize}
    \item \textbf{Does} stratify every one-dimensional marginal: each risk factor's random draws are forced to cover equal-probability bins, so tails are always visited
    \item \textbf{Does not} stratify the \emph{joint} distribution across all 39 factors and $\sim$17 time steps ($\sim$663 effective dimensions) -- interaction noise between factors remains
    \item \textbf{Unbiased:} each draw is still marginally uniform, mapped through the inverse normal CDF, so estimates stay unbiased
    \item \textbf{Cost:} draw generation grows with $N\times$dimension -- negligible next to repricing here
  \end{itemize}
  \vspace{6pt}
  This is exactly why Latin Hypercube helps a smooth average (EE) more reliably than a 99th-percentile tail (PFE99): the tail is set by interactions between a handful of extreme scenarios, which stratifying single marginals does not directly target.
\end{frame}

\partframe{V}{What the Engine Does, and What It Finds}
\section{Part V: Engine, Equations, and Headline Results}

\begin{frame}{The simulation pipeline}
  \begin{enumerate}
    \item \textbf{Calibrate:} pull real data $\to$ fit every model parameter (Part II)
    \item \textbf{Draw:} Latin Hypercube random numbers $\to$ correlated shocks (Cholesky decomposition of the real 39$\times$39 matrix)
    \item \textbf{Simulate:} step every risk factor jointly across the full time grid (equities, FX, USD rate, JPY rate)
    \item \textbf{Reprice:} feed simulated market states into the provided pricers -- every trade, every date, every scenario
    \item \textbf{Aggregate:} net by counterparty $\to$ EE / median PFE / PFE99 / MPE, on both exposure conventions
    \item \textbf{Downstream:} Greeks, stress, CVA/DVA/FVA/KVA, SA-CVA, SA-CCR -- all reusing the same simulated paths
  \end{enumerate}
\end{frame}

\begin{frame}{Key equations, exactly as implemented}
  \small
  \textbf{Rates (\texttt{models/rates.py}) -- Hull-White one-factor, shifted Ornstein-Uhlenbeck:}
  \[ dr(t) = [\theta(t) - a\,r(t)]\,dt + \sigma\,dW(t), \qquad r(t) = x(t) + \alpha(t) \]
  Exact-in-distribution step for $x(t)$; $\alpha(t)$ deterministic, fitted to today's curve; closed-form $P(t,T)$.

  \textbf{Equities and FX (\texttt{models/equity\_fx.py}) -- log-Euler GBM:}
  \[ d\ln S = (r_{USD} - q - \tfrac12\sigma^2)\,dt + \sigma\,dW \quad\text{(USD name)}; \qquad d\ln X = (r_{JPY}-r_{USD}+\tfrac12\sigma_X^2)\,dt + \sigma_X\,dW_X \]

  \textbf{Correlated shocks (\texttt{simulation/engine.py}):} $Z = L\,Z_{indep}$, Cholesky factor of the real 39$\times$39 historical matrix, applied every time step.

  \textbf{Exposure (\texttt{exposure/spec\_exposure.py}):} $\text{Exposure}_c(t) = \max\big(V_c(t{+}10\text{bd}) - V_c(t{-}1\text{bd}),\,0\big)$, book $= \sum_c \text{Exposure}_c$.

  \textbf{CVA (\texttt{exposure/cva.py}), Basel MAR50.32 discretisation:} $\text{CVA} = \text{LGD}\sum_i \text{DEE}(t_i)\cdot\text{PD}(t_{i-1},t_i)$.
\end{frame}

\begin{frame}{Method: common random numbers, and what they buy}
  \small
  \begin{itemize}
    \item Bumping a risk factor and repricing with \textbf{independent} random draws: estimator std.\ dev.\ \$245k -- too noisy to use
    \item Reusing the \textbf{identical} draws in the base and bumped run (common random numbers, CRN): std.\ dev.\ \$1k
    \item \textbf{282x noise reduction} -- this, not the sampling scheme, is what actually controls Greeks precision
  \end{itemize}
  \vspace{6pt}
  \centering

\begin{frame}{An exact rate Greek, reusing the same simulated draws}
  \small
  In Hull-White 1F, $r(t)=x(t)+\alpha(t)$: $x(t)$ is the simulated random factor (depends only on random draws, $a$, $\sigma$); $\alpha(t)$ is deterministic (depends only on the curve). A curve bump that leaves $a,\sigma$ fixed -- every DV01 bump here -- leaves $x(t)$ \textbf{identical} to the base run.
  \begin{itemize}
    \item The whole bump effect is one closed-form number per date, added to the base run's own paths
    \item \textbf{No resimulation. No new random draws.}
    \item Validated against a true independent resimulation: agreement to $\mathbf{10^{-15}}$ relative precision -- not an approximation, the same calculation
  \end{itemize}
  \vspace{6pt}
  \textbf{Separately: does Latin Hypercube itself reduce Greeks noise?} Tested directly -- mixed, 15--85\% less noise on EE deltas, no consistent benefit on PFE99 deltas:
