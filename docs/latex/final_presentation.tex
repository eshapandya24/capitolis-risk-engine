\section{Data: Sources, Uses, and Why Real Data Matters}

\begin{frame}{Data philosophy}
  \begin{itemize}
    \item \textbf{Every number in this project traces to a disclosed, real source} -- no synthetic placeholders for calibration inputs
    \item \textbf{Benefit:} results are reproducible, auditable, and defensible in front of a risk committee; every assumption is a named simplification, not a silent one
    \item Two tiers: \textbf{public} (Yahoo Finance, FRED, Databento) and \textbf{licensed} (Bloomberg, used and cross-checked but never redistributed as raw series)
  \end{itemize}

\section{Engine Architecture}

\begin{frame}{The simulation pipeline}
  \begin{enumerate}
    \item \textbf{Calibrate:} pull real data $\to$ fit every model parameter (previous section)
    \item \textbf{Draw:} Latin Hypercube random numbers $\to$ correlated shocks (Cholesky decomposition of the real 39$\times$39 matrix)
    \item \textbf{Simulate:} step every risk factor jointly across the full time grid (equities, FX, USD rate, JPY rate)
    \item \textbf{Reprice:} feed simulated market states into the provided pricers -- every trade, every date, every scenario
    \item \textbf{Aggregate:} net by counterparty $\to$ EE / median PFE / PFE99 / MPE, on both exposure conventions
    \item \textbf{Downstream:} Greeks, stress, CVA/DVA/FVA/KVA, SA-CVA, SA-CCR -- all reusing the same simulated paths
  \end{enumerate}
\end{frame}

\end{frame}

\section{Risk-Factor Models and Calibration}

\begin{frame}{Equity and FX: correlated geometric Brownian motion}
  \begin{columns}[T]
    \begin{column}{0.4\textwidth}
      \begin{itemize}
        \item Standard lognormal diffusion, risk-neutral drift ($r-q$), correlated via the real 39$\times$39 matrix
        \item JPY-quoted names carry a quanto correction; USDJPY drift enforces the martingale condition on the JPY money-market account valued in USD
      \end{itemize}
    \end{column}
    \begin{column}{0.6\textwidth}
      \includegraphics[width=\linewidth]{figures/fig13.pdf}
    \end{column}
  \end{columns}
  \small\centering Cholesky demonstration on the most correlated real pair (corr 0.82): independent draws (left) become correlated (right) after multiplying by $L$.
\end{frame}

\begin{frame}{Correlation: full Cholesky matrix vs.\ PCA factor model}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \includegraphics[width=\linewidth]{figures/fig14.pdf}
    \end{column}
    \begin{column}{0.5\textwidth}
      \small
      \begin{tabular}{p{0.24\textwidth}p{0.18\textwidth}p{0.18\textwidth}p{0.2\textwidth}}
        \toprule
        & \textbf{Full} & \textbf{PCA k=5} & \textbf{PCA k=10} \\

        Step time (5,000 scen.) & 1.91s & 2.08s & 2.29s \\

      \vspace{4pt}

    \end{column}

\end{frame}

\begin{frame}{USD rates: calibrating mean reversion ($a$) -- two real routes compared}
  \small
  \begin{tabular}{p{0.2\textwidth}p{0.27\textwidth}p{0.08\textwidth}p{0.08\textwidth}p{0.15\textwidth}}
    \toprule
    \textbf{Route} & \textbf{Data} & \textbf{$a$} & \textbf{$R^2$} & \textbf{Verdict} \\
    \midrule
    Swaption cube (preferred) & Bloomberg ATM normal vol, 1M expiry, 1--15Y tenors & 0.0167 & 0.845 & \good\ used \\
    SOFR-futures vol decay & $\sim$2y Databento history, 8 contracts & 0.0458 & 0.820 & cross-check only \\
    \bottomrule
  \end{tabular}
  \vspace{8pt}
  Both real-data routes give sensible, positive values with similar fit quality; the swaption route is the textbook standard (implied, not realised) and was preferred. The two disagree because they use different instruments (implied option vol vs.\ realised futures moves) and different tenor ranges -- a real, disclosed source of model uncertainty, quantified directly in the model-risk study ($a\times3$ moves the headline number by $-5\%$).
\end{frame}

\begin{frame}{USD rates: calibrating volatility ($\sigma$) -- realised, and which window}
  \small
  \begin{itemize}
    \item \textbf{Original source:} realised volatility of the overnight SOFR fixing itself, 0.63\%
    \item \textbf{Problem:} overnight fixings move in discrete steps on Fed days -- understates how far the \emph{long-dated} yields this book depends on actually move (realised 10-day change of the 20y yield: $\sim$15bp std dev, vs.\ $\sim$11bp implied by the overnight sigma)
    \item \textbf{\good\ Changed to:} realised volatility of 2y--30y Treasury yield changes directly, 0.96\%
  \end{itemize}
  \vspace{8pt}
  \textbf{Validated by Kupiec backtest (99\% VaR, rate moves, 161 ten-day windows, 3 tenors):}
  \small
  \begin{tabular}{lrrrr}
    \toprule
    \textbf{Tenor} & \textbf{Overnight-SOFR $\sigma$: up/down exc.} & \textbf{Long-end $\sigma$: up/down exc.} & \textbf{Expected} & \textbf{Both pass?} \\
    \midrule
    5y & 1 / 1 & 4 / 3 & 1.6 & yes (p$>$0.05) \\
    10y & 1 / 0 & 3 / 2 & 1.6 & yes \\
    20y & 1 / 1 & 3 / 1 & 1.6 & yes \\
    \bottomrule
  \end{tabular}
  \vspace{4pt}
  Both sigma choices pass Kupiec at this sample size; the switch to long-end realised vol is motivated by the direct realised-move comparison above, which is the more targeted diagnostic for this book's long-dated exposure.
\end{frame}

  \small

  \begin{tabular}{lrrr}
    \toprule
    \textbf{Method} & \textbf{RMSE} & \textbf{vs.\ pseudo-random} & \textbf{ms/trial} \\
    \midrule

\section{Sensitivities (Greeks)}

\begin{frame}{Method: common random numbers, and what they buy}
  \small
  \begin{itemize}
    \item Bumping a risk factor and repricing with \textbf{independent} random draws: estimator std.\ dev.\ \$245k -- too noisy to use
    \item Reusing the \textbf{identical} draws in the base and bumped run (common random numbers, CRN): std.\ dev.\ \$1k
    \item \textbf{282x noise reduction} -- this, not the sampling scheme, is what actually controls Greeks precision
  \end{itemize}
  \vspace{6pt}
  \centering
  \includegraphics[width=0.6\linewidth]{figures/fig_greeks_cost.pdf}
  \small\vspace{2pt} Equity/FX spot bumps rescale simulated GBM paths exactly (no resimulation): 78 bumps at N=300 cost 88s actual vs.\ $\sim$8,166s estimated naive resimulation -- \textbf{93x cheaper}.
\end{frame}

\begin{frame}{Method comparison: bump-and-reprice vs.\ pathwise vs.\ adjoint}
  \small
  \begin{tabular}{p{0.2\textwidth}p{0.22\textwidth}p{0.42\textwidth}}
    \toprule
    \textbf{Method} & \textbf{Cost} & \textbf{Verdict} \\
    \midrule
    Bump, independent draws & baseline, \$245k noise & \bad: noise swamps the signal \\

\begin{frame}{Rate DV01: why the new method, and what changed}
  \small
  \textbf{Old:} bump the \emph{fitted} zero curve at 8 hand-picked tenors, triangular interpolation -- disconnected from how the curve was actually built.

  \textbf{\good\ New:} bump the curve's own $\sim$45 native construction pillars (real SOFR-futures-implied points + every Bloomberg-spliced long-end point), grouped into the same 8 report buckets by nearest pillar -- the genuine par-instrument (Jacobian) sensitivity a rates desk would actually compute.
  \vspace{8pt}
  \textbf{Material benefit:} the old grid was smearing the book's real 20--50y rate risk (where BF\_0003, the dominant \$500M bond forward maturing 2049, actually sits) partly into the 10y bucket. The new method correctly shows \textbf{82\% of portfolio DV01 in the 30y bucket}, not 54\%. Both methods sum to the same total (internal consistency check, within 0.1\%) -- the total risk was always right, only the attribution across tenor was wrong.
\end{frame}

\begin{frame}{An exact rate Greek, reusing the same simulated draws}
  \small
  In one-factor Hull-White, $r(t)=x(t)+\alpha(t)$: $x(t)$ is the simulated random factor (depends only on random draws, $a$, $\sigma$); $\alpha(t)$ is deterministic (depends only on the curve). A curve bump that leaves $a,\sigma$ fixed -- every DV01 bump here -- leaves $x(t)$ \textbf{identical} to the base run.
  \begin{itemize}
    \item The whole bump effect is one closed-form number per date, added to the base run's own paths
    \item \textbf{No resimulation. No new random draws.}
    \item Validated against a true independent resimulation: agreement to $\mathbf{10^{-15}}$ relative precision -- not an approximation, the same calculation
  \end{itemize}
  \vspace{6pt}
  \textbf{Separately: does Latin Hypercube itself reduce Greeks noise?} Tested directly -- mixed, 15--85\% less noise on EE deltas, no consistent benefit on PFE99 deltas:

\end{frame}
